DROP TABLE IF EXISTS hens;
CREATE TABLE hens (
    id SERIAL PRIMARY KEY,
    arrival_date DATE NOT NULL,
    quantity INT NOT NULL CHECK (quantity > 0), -- Cantidad de gallinas
    status CHAR(1) DEFAULT 'A' -- Activo ('A') o Inactivo ('I')
);

DROP TABLE IF EXISTS cycle_life;
CREATE TABLE cycle_life (
    id SERIAL PRIMARY KEY,
    hen_id INT REFERENCES hens(id) ON DELETE CASCADE,
    type_ito VARCHAR(20) NOT NULL, 
    name_ito VARCHAR(200) NOT NULL, 
    type_time VARCHAR(15) NOT NULL, -- Día, Semana, etc.
    times INT NOT NULL, -- Cantidad de días o semanas
    end_date DATE NOT NULL,
    status CHAR(1) DEFAULT 'A' -- Activo ('A') o Inactivo ('I')
);

-- Eliminar el trigger si ya existe
DROP TRIGGER IF EXISTS set_end_date_trigger ON cycle_life;
DROP FUNCTION IF EXISTS set_end_date();

-- Crear la función para calcular end_date
CREATE OR REPLACE FUNCTION calculate_end_date()
RETURNS TRIGGER AS $$
DECLARE
    arrival_date DATE;
BEGIN
    -- Obtener la arrival_date de la tabla hens
    SELECT h.arrival_date INTO arrival_date FROM hens h WHERE h.id = NEW.hen_id;
    
    IF arrival_date IS NULL THEN
        RAISE EXCEPTION 'No se encontró la fecha de llegada para el lote con id: %', NEW.hen_id;
    END IF;
    
    -- Calcular end_date en función de type_time y times
    IF NEW.type_time = 'Día' THEN
        NEW.end_date := arrival_date + NEW.times;
    ELSIF NEW.type_time = 'Semana' THEN
        NEW.end_date := arrival_date + (NEW.times * 7);
    ELSE
        RAISE EXCEPTION 'Tipo de tiempo no válido: %', NEW.type_time;
    END IF;
    
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_calculate_end_date
BEFORE INSERT OR UPDATE ON cycle_life
FOR EACH ROW
EXECUTE FUNCTION calculate_end_date();

CREATE OR REPLACE VIEW view_cycle_life AS
SELECT 
    cl.id AS cycle_life_id,
    cl.hen_id,
    h.arrival_date,
    h.quantity,
    cl.type_ito,
    cl.name_ito,
    cl.type_time,
    cl.times,
    cl.end_date,
    cl.status
FROM cycle_life cl
JOIN hens h ON cl.hen_id = h.id;

SELECT * FROM view_cycle_life;

CREATE OR REPLACE FUNCTION update_end_date_on_hens_update()
RETURNS TRIGGER AS $$
BEGIN
    UPDATE cycle_life
    SET end_date = 
        CASE 
            WHEN type_time = 'Día' THEN NEW.arrival_date + times
            WHEN type_time = 'Semana' THEN NEW.arrival_date + (times * 7)
            ELSE end_date 
        END
    WHERE hen_id = NEW.id;
    
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_update_end_date_on_hens_update
AFTER UPDATE ON hens
FOR EACH ROW
EXECUTE FUNCTION update_end_date_on_hens_update();
